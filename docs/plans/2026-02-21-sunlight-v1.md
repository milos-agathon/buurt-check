**Goal:** Implement the full Sunlight v1 spec from `docs/sunlight-prd.md` — delivering clear-sky direct sun visibility hours, Sky View Factor (SVF) diffuse metric, shadow time slider, accumulated roof heatmap, and PRD-aligned UX copy with disclaimers.

**Architecture:** Client-side computation in Three.js. Enhanced CPU raycasting (Path B) with cooperative scheduling for multi-point roof sampling and numeric metrics. Existing shadow map for visual time slider. CubeCamera hemispherical rendering for SVF (averaged across 3-5 roof points). Vertex-color heatmap on roof mesh with material clone/restore. Backend already has scoring/comparison/viewing-question stubs ready to consume new data.

**Tech Stack:** Three.js (WebGLRenderer, Raycaster, CubeCamera, WebGLCubeRenderTarget), SunCalc, React 18, TypeScript 5, Vitest, existing Polar Frost CSS tokens.

**Key architecture decisions from adversarial review (rev 2):**
1. **Cooperative scheduling** for raycasting loop (setTimeout chunks, not blocking main thread) — PRD Section 3.3 requires this
2. **30-minute sampling intervals** using fractional sunrise/sunset hours — integer truncation loses ~2h/day
3. **Per-face cubemap readback** via `renderer.setRenderTarget(cubeTarget, faceIndex)` — reading without targeting face reads same data 5x
4. **Material clone before heatmap** — store original MeshStandardMaterial for toggle-off restoration
5. **SVF averaged across 3-5 points** — single-centroid is too weak a proxy for whole roof
6. **Shadow slider throttled to ~12fps** — RAF cadence causes 480 shadow map renders in 8s, drops frames on mobile
7. **Backend RiskLevel → SeverityLevel migration** — frontend sends `good/moderate/poor/critical`, backend `RiskLevel` enum only has `low/medium/high`
8. **Single SVF pixel-classification code path** — `computeSvf` delegates to `computeSvfFromPixels` (no inlined duplicate)
9. **Memoized baseDate in ShadowTimeSlider** — `getPresetDate()` returns new Date() each render, causing callback churn

```typescript
// sunPosition.ts
import SunCalc from 'suncalc';
import { Vector3 } from 'three';

/** Returns normalized sun direction vector, or null if sun below horizon. */
export function getSunDirection(date: Date, lat: number, lng: number): Vector3 | null {
  const pos = SunCalc.getPosition(date, lat, lng);
  if (pos.altitude <= 0) return null;

  const az = pos.azimuth;
  const alt = pos.altitude;
  return new Vector3(
    -Math.sin(az) * Math.cos(alt),
    Math.sin(alt),
    Math.cos(az) * Math.cos(alt),
  ).normalize();
}

/** Returns sunrise/sunset as fractional hours (e.g., 5.5 = 5:30). */
export function getDaylightRange(date: Date, lat: number, lng: number): { sunrise: number; sunset: number } {
  const times = SunCalc.getTimes(date, lat, lng);
  return {
    sunrise: times.sunrise.getHours() + times.sunrise.getMinutes() / 60,
    sunset: times.sunset.getHours() + times.sunset.getMinutes() / 60,
  };
}

/** Returns the 21st of each month for the given year (monthly representative). */
export function getRepresentativeDates(year: number): Date[] {
  return Array.from({ length: 12 }, (_, i) => new Date(year, i, 21));
}

/**
 * Radius for DirectionalLight positioning (meters from scene center).
 * NOT the actual sun distance — this is how far to place the light source
 * to cover the scene's shadow frustum (±300m). The viewer already defines
 * an identical constant — when wiring, import from here to deduplicate (M1).
 */
export const SUN_DISTANCE = 300;
```

```typescript
// ShadowTimeSlider.tsx
import { useState, useCallback, useRef, useEffect, useMemo } from 'react';
import { useTranslation } from 'react-i18next';
import { getDaylightRange } from '../utils/sunPosition';
import './ShadowTimeSlider.css';

interface Props {
  lat: number;
  lng: number;
  onChange: (date: Date) => void;
}

type DatePreset = 'winter' | 'equinox' | 'summer' | 'today';

function getPresetDate(preset: DatePreset): Date {
  const year = new Date().getFullYear();
  switch (preset) {
    case 'winter': return new Date(year, 11, 21);
    case 'equinox': return new Date(year, 2, 21);
    case 'summer': return new Date(year, 5, 21);
    case 'today': return new Date();
  }
}

export default function ShadowTimeSlider({ lat, lng, onChange }: Props) {
  const { t } = useTranslation();
  const [preset, setPreset] = useState<DatePreset>('summer');
  const [minutes, setMinutes] = useState(720); // noon default
  const [playing, setPlaying] = useState(false);
  const animRef = useRef<number>(0);

  // N3 fix: Memoize baseDate to avoid new Date() on every render,
  // which would cause emitChange callback churn and effect re-fires
  const baseDate = useMemo(() => getPresetDate(preset), [preset]);
  const daylight = getDaylightRange(baseDate, lat, lng);
  const minMinutes = Math.floor(daylight.sunrise * 60);
  const maxMinutes = Math.ceil(daylight.sunset * 60);

  const emitChange = useCallback((mins: number) => {
    const d = new Date(baseDate);
    d.setHours(Math.floor(mins / 60), mins % 60, 0, 0);
    onChange(d);
  }, [baseDate, onChange]);

  const handlePreset = (p: DatePreset) => {
    setPreset(p);
    setMinutes(720);
    emitChange(720);
  };

  const handleSlider = (e: React.ChangeEvent<HTMLInputElement>) => {
    const m = Number(e.target.value);
    setMinutes(m);
    emitChange(m);
  };

  // Playback animation — throttled to ~12fps to avoid 480 shadow map renders in 8s (D7 fix)
  useEffect(() => {
    if (!playing) return;
    let current = minMinutes;
    let lastTime = 0;
    const FRAME_INTERVAL = 83; // ~12fps (1000/12) — shadow map at 2048x2048 is expensive

    const step = (timestamp: number) => {
      if (timestamp - lastTime < FRAME_INTERVAL) {
        animRef.current = requestAnimationFrame(step);
        return;
      }
      lastTime = timestamp;
      current += 5; // 5 minutes per tick at 12fps ≈ day in ~10s
      if (current > maxMinutes) {
        setPlaying(false);
        return;
      }
      setMinutes(current);
      emitChange(current);
      animRef.current = requestAnimationFrame(step);
    };
    animRef.current = requestAnimationFrame(step);
    return () => cancelAnimationFrame(animRef.current);
  }, [playing, minMinutes, maxMinutes, emitChange]);

  const hours = Math.floor(minutes / 60);
  const mins = minutes % 60;
  const timeLabel = `${hours}:${String(mins).padStart(2, '0')}`;

  return (
    <div className="shadow-slider">
      <div className="shadow-slider__presets">
        {(['winter', 'equinox', 'summer', 'today'] as const).map(p => (
          <button
            key={p}
            className={`shadow-slider__preset ${preset === p ? 'shadow-slider__preset--active' : ''}`}
            onClick={() => handlePreset(p)}
          >
            {t(`shadow_slider.${p}`)}
          </button>
        ))}
      </div>
      <div className="shadow-slider__controls">
        <button
          className="shadow-slider__play"
          onClick={() => setPlaying(!playing)}
          aria-label={playing ? t('shadow_slider.pause') : t('shadow_slider.play')}
        >
          {playing ? '⏸' : '▶'}
        </button>
        <input
          type="range"
          className="shadow-slider__range"
          min={minMinutes}
          max={maxMinutes}
          value={minutes}
          onChange={handleSlider}
          aria-label={t('shadow_slider.time')}
        />
        <span className="shadow-slider__time">{timeLabel}</span>
      </div>
    </div>
  );
}
```

```typescript
/** Ray-casting point-in-polygon test (2D). */
export function isPointInPolygon(point: number[], polygon: number[][]): boolean {
  let inside = false;
  for (let i = 0, j = polygon.length - 1; i < polygon.length; j = i++) {
    const xi = polygon[i][0], yi = polygon[i][1];
    const xj = polygon[j][0], yj = polygon[j][1];
    if ((yi > point[1]) !== (yj > point[1]) &&
        point[0] < (xj - xi) * (point[1] - yi) / (yj - yi) + xi) {
      inside = !inside;
    }
  }
  return inside;
}

/**
 * Generate a regular grid of 3D evaluation points on the roof.
 * Returns array of [x, y, z] where y = roofHeight + 0.5m (clearance above roof).
 * x/z are in the local coordinate system matching the footprint offsets.
 *
 * Uses even-odd ray-casting rule for point-in-polygon test.
 */
export function generateRoofGrid(
  footprint: number[][],         // [[dx, dz], ...] offsets from center
  roofHeight: number,            // height in local coords (ground-relative)
  gridSpacing: number = 2.0,     // meters between sample points
  maxPoints: number = 100,       // cap to prevent performance issues
): [number, number, number][] {
  // Compute bounding box
  const xs = footprint.map(p => p[0]);
  const zs = footprint.map(p => p[1]);
  const minX = Math.min(...xs), maxX = Math.max(...xs);
  const minZ = Math.min(...zs), maxZ = Math.max(...zs);

  const evalY = roofHeight + 0.5; // 0.5m above roof
  const points: [number, number, number][] = [];

  // Regular grid
  for (let x = minX + gridSpacing / 2; x <= maxX; x += gridSpacing) {
    for (let z = minZ + gridSpacing / 2; z <= maxZ; z += gridSpacing) {
      if (isPointInPolygon([x, z], footprint)) {
        points.push([x, evalY, z]);
        if (points.length >= maxPoints) return points;
      }
    }
  }

  // Ensure at least centroid is included
  if (points.length === 0) {
    const cx = xs.reduce((a, b) => a + b, 0) / xs.length;
    const cz = zs.reduce((a, b) => a + b, 0) / zs.length;
    points.push([cx, evalY, cz]);
  }

  return points;
}
```

```typescript
// sunlightAnalysis.ts
import { Vector3, Raycaster } from 'three';
import type { Object3D } from 'three';
import SunCalc from 'suncalc';
import { SUN_DISTANCE } from './sunPosition';
import type { SunlightResult } from '../types/api';

export interface MultiPointSunlightResult {
  /** Per-point monthly hours: [pointIndex][monthIndex] */
  perPoint: number[][];
  /** Aggregated summary (mean across points) — same shape as SunlightResult */
  summary: SunlightResult;
  /** Per-point annual average hours */
  perPointAnnual: number[];
}

interface AnalysisInput {
  evaluationPoints: [number, number, number][];
  buildingMeshes: Object3D[];
  targetPandId: string;
  lat: number;
  lng: number;
  year: number;
}

const WINTER_IDX = 11;
const EQUINOX_IDX = 2;
const SUMMER_IDX = 5;

/**
 * IMPORTANT: This function must NOT block the main thread for >50ms.
 * Use computeMultiPointSunlightAsync() in the viewer, which wraps this
 * with cooperative scheduling (setTimeout chunks).
 */
export function computeMultiPointSunlight(input: AnalysisInput): MultiPointSunlightResult {
  const { evaluationPoints, buildingMeshes, targetPandId, lat, lng, year } = input;
  const raycaster = new Raycaster();
  const monthlyDates = Array.from({ length: 12 }, (_, i) => new Date(year, i, 21));

  // perPoint[pointIdx][monthIdx] = hours
  const perPoint: number[][] = evaluationPoints.map(() => []);

  const STEP_MINUTES = 30; // 30-minute intervals (not hourly) — C2 fix

  for (let m = 0; m < monthlyDates.length; m++) {
    const date = monthlyDates[m];
    const times = SunCalc.getTimes(date, lat, lng);
    // Use fractional hours — integer truncation loses ~2h/day (C2 fix)
    const sunriseMin = Math.floor((times.sunrise.getHours() + times.sunrise.getMinutes() / 60) * 60);
    const sunsetMin = Math.ceil((times.sunset.getHours() + times.sunset.getMinutes() / 60) * 60);

    for (let ptIdx = 0; ptIdx < evaluationPoints.length; ptIdx++) {
      const [px, py, pz] = evaluationPoints[ptIdx];
      const origin = new Vector3(px, py, pz);
      let sunlitSteps = 0;
      let totalSteps = 0;

      for (let min = sunriseMin; min <= sunsetMin; min += STEP_MINUTES) {
        const d = new Date(date);
        d.setHours(Math.floor(min / 60), min % 60, 0, 0);
        const sunPos = SunCalc.getPosition(d, lat, lng);
        if (sunPos.altitude <= 0) continue;

        totalSteps++;
        const az = sunPos.azimuth;
        const alt = sunPos.altitude;
        const sunDir = new Vector3(
          -Math.sin(az) * Math.cos(alt),
          Math.sin(alt),
          Math.cos(az) * Math.cos(alt),
        ).normalize();

        raycaster.set(origin, sunDir);
        raycaster.far = SUN_DISTANCE * 2;

        const hits = raycaster.intersectObjects(buildingMeshes);
        const blocked = hits.some(
          (hit) => hit.object.userData.pandId !== targetPandId
                && !hit.object.userData.isGround,
        );

        if (!blocked) sunlitSteps++;
      }

      // Convert steps to hours: each step = 0.5h
      const sunlitHours = sunlitSteps * (STEP_MINUTES / 60);
      perPoint[ptIdx].push(Math.round(sunlitHours * 10) / 10);
    }
  }

  // Per-point annual averages
  const perPointAnnual = perPoint.map(
    months => Math.round((months.reduce((a, b) => a + b, 0) / 12) * 10) / 10
  );

  // Aggregated summary: mean across all points per month
  const meanMonthly = Array.from({ length: 12 }, (_, m) => {
    const sum = perPoint.reduce((acc, pt) => acc + pt[m], 0);
    return Math.round((sum / evaluationPoints.length) * 10) / 10;
  });

  return {
    perPoint,
    perPointAnnual,
    summary: {
      winter: meanMonthly[WINTER_IDX],
      equinox: meanMonthly[EQUINOX_IDX],
      summer: meanMonthly[SUMMER_IDX],
      annualAverage: Math.round((meanMonthly.reduce((a, b) => a + b, 0) / 12) * 10) / 10,
      analysisYear: year,
    },
  };
}
```

```typescript
// heatmapColors.ts

/**
 * Maps a value to RGB color on a red→yellow→green gradient.
 * Returns [r, g, b] each in [0, 1].
 */
export function sunHoursToColor(
  value: number,
  min: number,
  max: number,
): [number, number, number] {
  const t = Math.max(0, Math.min(1, (value - min) / (max - min || 1)));

  // Red → Yellow → Green gradient
  if (t < 0.5) {
    // Red to Yellow (r stays 1, g ramps up)
    return [0.9, t * 2 * 0.85, 0.1];
  } else {
    // Yellow to Green (r ramps down, g stays high)
    return [0.9 - (t - 0.5) * 2 * 0.7, 0.75 + (t - 0.5) * 0.1, 0.15 + (t - 0.5) * 0.2];
  }
}
```

```typescript
// In NeighborhoodViewer3D.tsx — new function
import { sunHoursToColor } from '../utils/heatmapColors';
import { BufferAttribute, Color } from 'three';

function applyRoofHeatmap(
  targetMesh: Mesh,
  roofGridPoints: [number, number, number][],
  perPointAnnual: number[],
): void {
  const geometry = targetMesh.geometry;
  const positions = geometry.getAttribute('position');
  const count = positions.count;

  // Compute min/max for color scale
  const minH = Math.min(...perPointAnnual);
  const maxH = Math.max(...perPointAnnual);

  // Assign vertex colors by nearest evaluation point
  const colors = new Float32Array(count * 3);
  for (let i = 0; i < count; i++) {
    const vx = positions.getX(i);
    const vz = positions.getZ(i); // Z in Three.js = north/south

    // Find nearest grid point (2D distance, ignore height)
    let nearestIdx = 0;
    let nearestDist = Infinity;
    for (let j = 0; j < roofGridPoints.length; j++) {
      const [gx, , gz] = roofGridPoints[j];
      const d = (vx - gx) ** 2 + (vz - gz) ** 2;
      if (d < nearestDist) {
        nearestDist = d;
        nearestIdx = j;
      }
    }

    const [r, g, b] = sunHoursToColor(perPointAnnual[nearestIdx], minH, maxH);
    colors[i * 3] = r;
    colors[i * 3 + 1] = g;
    colors[i * 3 + 2] = b;
  }

  geometry.setAttribute('color', new BufferAttribute(colors, 3));
  // Switch material to vertex colors mode
  const mat = targetMesh.material as MeshStandardMaterial;
  mat.vertexColors = true;
  mat.needsUpdate = true;
}
```

**Step 2: Implement SVF computation**

```typescript
// svfComputation.ts
import {
  WebGLCubeRenderTarget,
  CubeCamera,
  Scene,
  Color,
  Vector3,
  MeshBasicMaterial,
  Mesh,
} from 'three';
import type { WebGLRenderer, Object3D, BufferGeometry } from 'three';

const SKY_COLOR: [number, number, number] = [135, 206, 235]; // Distinct sky blue
const OBSTRUCTION_COLOR = 0x000000; // Black for buildings
const CUBE_SIZE = 64; // 64x64 per face — balance between accuracy and performance
const SKY_TOLERANCE = 30; // Color matching tolerance

/**
 * Cosine weight for a pixel on a cubemap face using proper solid angle.
 *
 * For SVF, we need the cosine of the zenith angle (angle from straight up).
 * Each cubemap pixel maps to a direction vector. The cosine weight is the
 * Y-component of the normalized direction (since Y = up in our scene).
 *
 * The solid angle subtended by each pixel also varies across the face —
 * edge pixels cover less solid angle than center pixels. The differential
 * solid angle for a cubemap pixel at (u,v) on a unit cube is:
 *   dΩ = 1 / (u² + v² + 1)^(3/2)
 *
 * D2 fix: Use direction-vector approach instead of ad-hoc face type classification.
 */
export function cosineWeightForCubemapPixel(
  faceIndex: number,
  px: number,
  py: number,
  size: number,
): number {
  // Normalize pixel to [-1, 1] on the face
  const u = (px + 0.5) / size * 2 - 1;
  const v = -((py + 0.5) / size * 2 - 1); // Flip V (cubemap convention)

  // Map (faceIndex, u, v) to a 3D direction vector
  // Three.js cubemap face order: +X, -X, +Y, -Y, +Z, -Z
  let dx: number, dy: number, dz: number;
  switch (faceIndex) {
    case 0: dx =  1; dy =  v; dz = -u; break; // +X
    case 1: dx = -1; dy =  v; dz =  u; break; // -X
    case 2: dx =  u; dy =  1; dz = -v; break; // +Y (top)
    case 3: dx =  u; dy = -1; dz =  v; break; // -Y (bottom)
    case 4: dx =  u; dy =  v; dz =  1; break; // +Z
    case 5: dx = -u; dy =  v; dz = -1; break; // -Z
    default: return 0;
  }

  // Normalize direction
  const len = Math.sqrt(dx * dx + dy * dy + dz * dz);
  const ny = dy / len; // Y component = cos(zenith angle from up)

  // Only upper hemisphere contributes (Y > 0)
  if (ny <= 0) return 0;

  // Weight = cos(zenith) * dΩ, where dΩ = 1/(u²+v²+1)^(3/2)
  const r2 = u * u + v * v + 1;
  const solidAngle = 1 / (r2 * Math.sqrt(r2));

  return ny * solidAngle;
}

/**
 * Classify pixels as sky or obstruction and compute weighted SVF for one cubemap face.
 * Uses proper solid-angle integration with cosine weighting (D2 fix).
 */
export function computeSvfFromPixels(
  pixels: Uint8Array,
  size: number,
  skyColor: number[],
  faceIndex: number,
): number {
  let weightedSky = 0;
  let totalWeight = 0;

  for (let y = 0; y < size; y++) {
    for (let x = 0; x < size; x++) {
      const idx = (y * size + x) * 4;
      const r = pixels[idx];
      const g = pixels[idx + 1];
      const b = pixels[idx + 2];

      const weight = cosineWeightForCubemapPixel(faceIndex, x, y, size);
      if (weight <= 0) continue;

      totalWeight += weight;

      // Check if pixel matches sky color (within tolerance)
      const dr = Math.abs(r - skyColor[0]);
      const dg = Math.abs(g - skyColor[1]);
      const db = Math.abs(b - skyColor[2]);
      if (dr < SKY_TOLERANCE && dg < SKY_TOLERANCE && db < SKY_TOLERANCE) {
        weightedSky += weight;
      }
    }
  }

  return totalWeight > 0 ? weightedSky / totalWeight : 0;
}

/**
 * Compute SVF at a single evaluation point using hemispherical cubemap rendering.
 *
 * @param renderer - Active WebGLRenderer
 * @param buildingGeometries - Array of {geometry, position} for scene buildings
 * @param evalPoint - [x, y, z] evaluation point in scene coords
 * @returns SVF value in [0, 1] where 0 = fully enclosed, 1 = open sky
 */
export function computeSvf(
  renderer: WebGLRenderer,
  buildingMeshes: Object3D[],
  evalPoint: [number, number, number],
): number {
  // Create a temporary scene with flat-colored buildings
  const svfScene = new Scene();
  svfScene.background = new Color(
    SKY_COLOR[0] / 255,
    SKY_COLOR[1] / 255,
    SKY_COLOR[2] / 255,
  );

  // Clone building meshes with flat obstruction material
  const obstructionMat = new MeshBasicMaterial({ color: OBSTRUCTION_COLOR });
  for (const mesh of buildingMeshes) {
    if (mesh instanceof Mesh) {
      const clone = new Mesh(mesh.geometry, obstructionMat);
      clone.position.copy(mesh.position);
      clone.rotation.copy(mesh.rotation);
      clone.scale.copy(mesh.scale);
      svfScene.add(clone);
    }
  }

  // Set up cube camera at evaluation point
  const cubeTarget = new WebGLCubeRenderTarget(CUBE_SIZE);
  const cubeCamera = new CubeCamera(0.1, 1000, cubeTarget);
  cubeCamera.position.set(evalPoint[0], evalPoint[1], evalPoint[2]);

  // Render cubemap
  cubeCamera.update(renderer, svfScene);

  // Read back pixels from each face and compute weighted SVF
  // C1 FIX: Must call setRenderTarget(cubeTarget, faceIndex) before readRenderTargetPixels
  // Without this, the same face data is read 5 times — producing wrong SVF.
  //
  // N2 FIX: Delegate to computeSvfFromPixels instead of inlining the pixel loop.
  // This keeps a single code path for pixel classification + weighting logic,
  // reducing maintenance risk. computeSvfFromPixels is also exported for unit testing.
  let weightedSky = 0;
  let totalWeight = 0;

  // The 6 cube faces: +X(0), -X(1), +Y(2), -Y(3), +Z(4), -Z(5)
  // Skip face 3 (-Y / nadir) — below ground, no sky contribution
  const UPPER_FACES = [0, 1, 2, 4, 5];

  for (const faceIndex of UPPER_FACES) {
    const pixels = new Uint8Array(CUBE_SIZE * CUBE_SIZE * 4);

    // CRITICAL: Target the specific cube face before reading pixels
    renderer.setRenderTarget(cubeTarget, faceIndex);
    renderer.readRenderTargetPixels(cubeTarget, 0, 0, CUBE_SIZE, CUBE_SIZE, pixels);

    // Delegate to shared pixel classification function (single code path — N2 fix)
    // computeSvfFromPixels returns face-level SVF ratio; accumulate into global sum
    const faceSvf = computeSvfFromPixels(pixels, CUBE_SIZE, SKY_COLOR, faceIndex);

    // Compute total weight for this face to merge into global average
    let faceWeight = 0;
    for (let y = 0; y < CUBE_SIZE; y++) {
      for (let x = 0; x < CUBE_SIZE; x++) {
        const w = cosineWeightForCubemapPixel(faceIndex, x, y, CUBE_SIZE);
        if (w > 0) faceWeight += w;
      }
    }
    weightedSky += faceSvf * faceWeight;
    totalWeight += faceWeight;
  }

  // Restore default render target
  renderer.setRenderTarget(null);

  // Cleanup
  obstructionMat.dispose();
  cubeTarget.dispose();

  return totalWeight > 0 ? weightedSky / totalWeight : 0;
}

/**
 * Compute SVF averaged across multiple roof points (D1 fix).
 * Single-centroid SVF is a weak proxy — averaging 3-5 points gives
 * a more representative value for the whole roof.
 */
export function computeSvfMultiPoint(
  renderer: WebGLRenderer,
  buildingMeshes: Object3D[],
  evalPoints: [number, number, number][],
  maxPoints: number = 5,
): number {
  // Sample up to maxPoints evenly distributed from the grid
  const step = Math.max(1, Math.floor(evalPoints.length / maxPoints));
  const sampled = evalPoints.filter((_, i) => i % step === 0).slice(0, maxPoints);

  const svfValues = sampled.map(pt => computeSvf(renderer, buildingMeshes, pt));
  return svfValues.reduce((a, b) => a + b, 0) / svfValues.length;
}
```

```typescript
        expect(cosineWeightForCubemapPixel(3, x, y, 64)).toBe(0);
      }
    }
  });

  it('all-sky scene returns SVF close to 1.0', () => {
    const skyColor = [135, 206, 235];
    const pixels = new Uint8Array(64 * 64 * 4);
    for (let i = 0; i < 64 * 64; i++) {
      pixels[i * 4] = 135;
      pixels[i * 4 + 1] = 206;
      pixels[i * 4 + 2] = 235;
      pixels[i * 4 + 3] = 255;
    }
    // Top face (index 2): all sky
    const svf = computeSvfFromPixels(pixels, 64, skyColor, 2);
    expect(svf).toBeGreaterThan(0.9);
  });

  it('half-obstructed side face returns intermediate SVF', () => {
    const skyColor = [135, 206, 235];
    const pixels = new Uint8Array(64 * 64 * 4);
    for (let y = 0; y < 64; y++) {
      for (let x = 0; x < 64; x++) {
        const idx = (y * 64 + x) * 4;
        if (y < 32) {
          // Upper half: sky
          pixels[idx] = 135; pixels[idx+1] = 206; pixels[idx+2] = 235;
        } else {
          // Lower half: obstruction
          pixels[idx] = 0; pixels[idx+1] = 0; pixels[idx+2] = 0;
        }
        pixels[idx+3] = 255;
      }
    }
    const svf = computeSvfFromPixels(pixels, 64, skyColor, 0); // +X face
    expect(svf).toBeGreaterThan(0.0);
    expect(svf).toBeLessThan(1.0);
  });
});
```
